- export const MAX*COPY_BYTES = 8 * 1024 \_ 1024
- export type DeliveryErrorCode = "SOURCE_MISSING" | "SOURCE_NOT_OWNED" | "CONTENT_TOO_LARGE"
- export type PlannedNote = { src: Doc<"notes">; parentIdx: number | null }
- export async function planCopy(ctx: QueryCtx | MutationCtx, rootId: Id<"notes">, opts: { requiredOwner?: Id<"users">; maxNotes?: number; maxBytes?: number }): Promise<{ ok: true; plan: PlannedNote[] } | { ok: false; error: DeliveryErrorCode }>
  requiredOwner set => root.owner must equal it (else SOURCE_NOT_OWNED) and descendants with another owner are skipped with their subtrees.
  maxNotes undefined => no cap (children read with collect, preserving duplicateNote behaviour); defined => children read with take(maxNotes + 1).
  maxBytes undefined => no byte budget.
- export async function writeCopy(ctx: MutationCtx, plan: PlannedNote[], opts: { owner: Id<"users">; parentNote?: Id<"notes">; rootTitle?: string; purchaseId?: Id<"purchases"> }): Promise<Id<"notes">>
  Returns the new root id. Does NOT touch opts.parentNote's childNotes (caller's job).
- export async function deliverPurchase(ctx: MutationCtx, purchaseId: Id<"purchases">): Promise<{ status: "delivered" | "failed" | "skipped"; noteId?: Id<"notes">; error?: DeliveryErrorCode }>
  "skipped" = purchase missing, not paid, or already delivered.
